#include <Arduino.h>
#include <FastAccelStepper.h>

#define ENCODER_A 34
#define ENCODER_B 35
#define STEP_PIN 26
#define DIR_PIN 27
#define ENABLE_PIN 14
#define HOME_PIN 25
#define END_PIN 33

volatile long encoderCount = 0;

const float Kp = 1000.0f;
const float Kd = 90.0f;
const float STEPS_PER_CM = 500.0f;
const long CART_LIMIT = 15000;

FastAccelStepperEngine engine;
FastAccelStepper* stepper = nullptr;

void IRAM_ATTR encoderA_ISR(){
  if (gpio_get_level((gpio_num_t) ENCODER_B) == 0){
        encoderCount++;
  } else {
        encoderCount--;
    }
}

void IRAM_ATTR encoderB_ISR() {
  if (gpio_get_level((gpio_num_t) ENCODER_A) == 0){
      encoderCount--;
    } else {
       encoderCount++;
    }
}

long zeroOffset = 0;

void home() {

    Serial.println("Homing...");

    stepper->setSpeedInHz(1000);

    stepper->runBackward();

    while (digitalRead(HOME_PIN) == HIGH) {
        delay(1);
    }

    stepper->forceStop();

    delay(100);

    stepper->setCurrentPosition(0);

    Serial.println("Home found");
}

long measureRail() {

    Serial.println("Measuring rail...");

    stepper->setSpeedInHz(1000);

    stepper->runForward();

    while (digitalRead(END_PIN) == HIGH) {
        delay(1);
    }

    stepper->forceStop();

    delay(100);

    long railLength = stepper->getCurrentPosition();

    Serial.print("Rail length = ");
    Serial.println(railLength);

    return railLength;
}

void moveToCenter(long railLegth){
  long center = railLegth / 2;

  stepper->moveTo(center);
  while(stepper->isRunning()){

  }
  stepper->setCurrentPosition(0);
}

void setup(){
  Serial.begin(115200);
  pinMode(ENCODER_A, INPUT);
  pinMode(ENCODER_B, INPUT);
  pinMode(HOME_PIN, INPUT);
  pinMode(END_PIN, INPUT);

  pinMode(ENABLE_PIN, OUTPUT);
  digitalWrite(ENABLE_PIN, LOW);


  engine.init();

  stepper = engine.stepperConnectToPin(STEP_PIN);

  if(stepper){
    stepper->setDirectionPin(DIR_PIN);
    stepper->setEnablePin(ENABLE_PIN);
    stepper->setAutoEnable(false);
    stepper->setSpeedInHz(2000); //    2000/1600 * 60 = 75 RPM maks
    stepper->setAcceleration(150000);//   4000/1600 * 60 = 3750 RPM/s maks accel
    stepper->setCurrentPosition(0);
  }

  home();
  long railLength = measureRail();
  moveToCenter(railLength);

  long leftLimit = -railLength / 2;
  long rightLimit = railLength / 2;

  attachInterrupt(digitalPinToInterrupt(ENCODER_A), encoderA_ISR, RISING);
  attachInterrupt(digitalPinToInterrupt(ENCODER_B), encoderB_ISR, RISING);
  
  delay(100);

  //Current position becomes 0
  zeroOffset = encoderCount;
}

void loop() {

  bool homeHit = digitalRead(HOME_PIN) == LOW;
  bool endHit  = digitalRead(END_PIN) == LOW;

  long counts = encoderCount - zeroOffset;

  float angleRaw = counts * (360.0f / 800.0f);

  float angle = angleRaw;

while (angle > 180.0f){
  angle -= 360.0f;
}


  while (angle < -180.0f){
    angle += 360.0f;
  }


  //Velocity calculation
  static float prevAngleRaw = angleRaw;
  static uint32_t lastMicros = micros();

  uint32_t nowMicros = micros();
  float dt = (nowMicros - lastMicros) * 1e-6f;

  float angleVel = 0;

  if (dt > 0)
      angleVel = (angleRaw - prevAngleRaw) / dt;

  //Timestamp for logging
  uint32_t nowMillis = millis();


  //PID tuning

float theta = angle - 180.0f;

while(theta > 180) theta -= 360;
while(theta < -180) theta += 360;

float thetaDot = angleVel;

float x = stepper->getCurrentPosition() / STEPS_PER_CM;

static float prevX = x;
float xVel = (x - prevX) / dt;
prevX = x;

const float Ktheta = 1200.0f;
const float KthetaDot = 80.0f;
const float Kx = 20.0f;
const float KxDot = 10.0f;

float cmd = Ktheta * theta + KthetaDot * thetaDot + Kx * x + KxDot * xVel;
  cmd = -cmd;
  //stepper driver
  int speed = abs(cmd);

if (homeHit && cmd > 0) {
    cmd = 0;
}

if (endHit && cmd < 0) {
    cmd = 0;
}

  if(speed < 20){
    stepper->stopMove();
  }
  else{
    stepper->setSpeedInHz(speed);
    if(cmd > 0){
      stepper->runBackward();
    }
    else{
      stepper->runForward();
    }
  }
  static uint32_t lastPrint = 0;

if (millis() - lastPrint > 20) {   // 50 Hz logging
    Serial.print(angle);
    Serial.print(",");

    Serial.print(angleVel);
    Serial.print(",");

    Serial.print(cmd);
    Serial.print(",");

    Serial.print(speed);
    Serial.print(",");

    lastPrint = millis();
}
static uint32_t prevLoop = micros();

  prevAngleRaw = angleRaw;
  lastMicros = nowMicros;
}
